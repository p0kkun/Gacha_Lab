### 改善案：API ルートでの認証チェック

クライアント側の認証だけでは不十分なため、API ルートでも認証をチェックします。

```typescript
// lib/admin-auth.ts
export function verifyAdminAuth(request: NextRequest): boolean {
  // セッションストレージはサーバー側でアクセスできないため、
  // クッキーまたはヘッダーで認証情報を送信する必要がある
  
  // 方法1: クッキーを使用
  const authToken = request.cookies.get('admin_auth_token')?.value;
  const expectedToken = process.env.ADMIN_AUTH_TOKEN; // サーバー側の環境変数
  
  return authToken === expectedToken;
}

// 方法2: 認証ミドルウェアを作成
export function withAdminAuth(handler: (req: NextRequest) => Promise<NextResponse>) {
  return async (req: NextRequest) => {
    if (!verifyAdminAuth(req)) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    return handler(req);
  };
}
```

### 1. ユーザー管理 (`/admin/users`)

#### 機能要件

- ユーザー一覧の表示（ページネーション）
- ユーザー詳細の表示
- ユーザーのガチャ実行履歴の表示
- ユーザー検索（ユーザーID、表示名）

#### API 実装例

```typescript
// app/api/admin/users/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { verifyAdminAuth } from '@/lib/admin-auth';

export async function GET(request: NextRequest) {
  if (!verifyAdminAuth(request)) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const { searchParams } = new URL(request.url);
  const page = parseInt(searchParams.get('page') || '1');
  const limit = parseInt(searchParams.get('limit') || '20');
  const search = searchParams.get('search') || '';

  const skip = (page - 1) * limit;

  const where = search
    ? {
        OR: [
          { userId: { contains: search } },
          { displayName: { contains: search } },
        ],
      }
    : {};

  const [users, total] = await Promise.all([
    prisma.user.findMany({
      where,
      skip,
      take: limit,
      orderBy: { createdAt: 'desc' },
      include: {
        _count: {
          select: {
            gachaHistories: true,
          },
        },
      },
    }),
    prisma.user.count({ where }),
  ]);

  return NextResponse.json({
    users,
    pagination: {
      page,
      limit,
      total,
      totalPages: Math.ceil(total / limit),
    },
  });
}
```

#### 機能要件

- アイテム一覧の表示（レアリティ別フィルタリング）
- アイテムの追加・編集・削除（論理削除）

- 有効/無効の切り替え

#### API 実装例

```typescript
// app/api/admin/items/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { verifyAdminAuth } from '@/lib/admin-auth';

export async function GET(request: NextRequest) {
  if (!verifyAdminAuth(request)) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const { searchParams } = new URL(request.url);
  const rarity = searchParams.get('rarity');
  const isActive = searchParams.get('isActive');

  const where: any = {};
  if (rarity) {
    where.rarity = rarity;
  }
  if (isActive !== null) {
    where.isActive = isActive === 'true';
  }

  const items = await prisma.gachaItem.findMany({
    where,
    orderBy: { createdAt: 'desc' },
  });

  return NextResponse.json({ items });
}

export async function POST(request: NextRequest) {
  if (!verifyAdminAuth(request)) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const body = await request.json();
  const { name, rarity, videoUrl, isActive } = body;

  const item = await prisma.gachaItem.create({
    data: {
      name,
      rarity,
      videoUrl,
      isActive: isActive ?? true,
    },
  });

  return NextResponse.json({ item });
}
```

### 4. ガチャ購入状況 (`/admin/statistics`)

#### 機能要件

- ガチャごとの課金人数・課金金額の表示
- 日別・月別の集計
- レアリティ別の排出率
- グラフ表示（Chart.js や Recharts を使用）

#### API 実装例

```typescript
// app/api/admin/statistics/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { verifyAdminAuth } from '@/lib/admin-auth';

export async function GET(request: NextRequest) {
  if (!verifyAdminAuth(request)) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const { searchParams } = new URL(request.url);
  const period = searchParams.get('period') || 'month'; // 'day' | 'month'

  // ガチャごとの統計
  const gachaStats = await prisma.gachaHistory.groupBy({
    by: ['gachaTypeId'],
    _count: {
      id: true,
    },
    where: {
      createdAt: {
        gte: getPeriodStartDate(period),
      },
    },
  });

  // レアリティ別の統計
  const rarityStats = await prisma.gachaHistory.groupBy({
    by: ['itemId'],
    _count: {
      id: true,
    },
    include: {
      item: {
        select: {
          rarity: true,
        },
      },
    },
  });

  // 日別/月別の集計
  const dailyStats = await prisma.$queryRaw`
    SELECT
      DATE(created_at) as date,
      COUNT(*) as count
    FROM gacha_histories
    WHERE created_at >= ${getPeriodStartDate(period)}
    GROUP BY DATE(created_at)
    ORDER BY date DESC
  `;

  return NextResponse.json({
    gachaStats,
    rarityStats,
    dailyStats,
  });
}

function getPeriodStartDate(period: string): Date {
  const now = new Date();
  if (period === 'day') {
    return new Date(now.getFullYear(), now.getMonth(), now.getDate());
  } else {
    return new Date(now.getFullYear(), now.getMonth(), 1);
  }
}
```

### 5. ガチャシミュレータ (`/admin/simulator`)

#### 機能要件

- 指定した回数（例: 10,000 回）のガチャをシミュレート
- 排出率の確認
- レアリティ別の実際の排出数と設定確率の比較
- グラフ表示

#### API 実装例

```typescript
// app/api/admin/simulator/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { verifyAdminAuth } from '@/lib/admin-auth';
import { prisma } from '@/lib/prisma';

export async function POST(request: NextRequest) {
  if (!verifyAdminAuth(request)) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const body = await request.json();
  const { gachaTypeId, iterations = 10000 } = body;

  const gachaType = await prisma.gachaType.findUnique({
    where: { id: gachaTypeId },
  });

  if (!gachaType) {
    return NextResponse.json({ error: 'Gacha type not found' }, { status: 404 });
  }

  // 重みの合計を計算
  const totalWeight =
    gachaType.firstPrizeWeight +
    gachaType.secondPrizeWeight +
    gachaType.thirdPrizeWeight +
    gachaType.fourthPrizeWeight +
    gachaType.fifthPrizeWeight +
    gachaType.loserWeight;

  // シミュレーション実行
  const results = {
    FIRST_PRIZE: 0,
    SECOND_PRIZE: 0,
    THIRD_PRIZE: 0,
    FOURTH_PRIZE: 0,
    FIFTH_PRIZE: 0,
    LOSER: 0,
  };

  for (let i = 0; i < iterations; i++) {
    const random = Math.random() * totalWeight;
    let cumulative = 0;

    cumulative += gachaType.firstPrizeWeight;
    if (random < cumulative) {
      results.FIRST_PRIZE++;
      continue;
    }

    cumulative += gachaType.secondPrizeWeight;
    if (random < cumulative) {
      results.SECOND_PRIZE++;
      continue;
    }

    cumulative += gachaType.thirdPrizeWeight;
    if (random < cumulative) {
      results.THIRD_PRIZE++;
      continue;
    }

    cumulative += gachaType.fourthPrizeWeight;
    if (random < cumulative) {
      results.FOURTH_PRIZE++;
      continue;
    }

    cumulative += gachaType.fifthPrizeWeight;
    if (random < cumulative) {
      results.FIFTH_PRIZE++;
      continue;
    }

    results.LOSER++;
  }

  // 実際の排出率を計算
  const actualRates = {
    FIRST_PRIZE: (results.FIRST_PRIZE / iterations) * 100,
    SECOND_PRIZE: (results.SECOND_PRIZE / iterations) * 100,
    THIRD_PRIZE: (results.THIRD_PRIZE / iterations) * 100,
    FOURTH_PRIZE: (results.FOURTH_PRIZE / iterations) * 100,
    FIFTH_PRIZE: (results.FIFTH_PRIZE / iterations) * 100,
    LOSER: (results.LOSER / iterations) * 100,
  };

  // 設定確率
  const expectedRates = {
    FIRST_PRIZE: (gachaType.firstPrizeWeight / totalWeight) * 100,
    SECOND_PRIZE: (gachaType.secondPrizeWeight / totalWeight) * 100,
    THIRD_PRIZE: (gachaType.thirdPrizeWeight / totalWeight) * 100,
    FOURTH_PRIZE: (gachaType.fourthPrizeWeight / totalWeight) * 100,
    FIFTH_PRIZE: (gachaType.fifthPrizeWeight / totalWeight) * 100,
    LOSER: (gachaType.loserWeight / totalWeight) * 100,
  };

  return NextResponse.json({
    iterations,
    results,
    actualRates,
    expectedRates,
  });
}
```
